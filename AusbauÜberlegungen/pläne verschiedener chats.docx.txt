```tsx
  onComplete 
}) => {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const [isDrawing, setIsDrawing] = useState(false);
  const [isRevealed, setIsRevealed] = useState(false);
  const contextRef = useRef<CanvasRenderingContext2D | null>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    // 1. Goldfolie zeichnen (Der "Rubbel"-Look)
    ctx.fillStyle = '#d4af37'; // Goldfarbe
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    
    // Optional: Text auf die Folie schreiben
    ctx.fillStyle = '#b8860b';
    ctx.font = 'bold 24px Arial';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText('✨ HIER RUBBELN ✨', canvas.width / 2, canvas.height / 2);

    contextRef.current = ctx;
  }, [width, height]);

  const getCoords = (e: React.MouseEvent | React.TouchEvent) => {
    const canvas = canvasRef.current;
    if (!canvas) return { x: 0, y: 0 };
    const rect = canvas.getBoundingClientRect();
    
    if ('touches' in e) {
      return {
        x: e.touches[0].clientX - rect.left,
        y: e.touches[0].clientY - rect.top,
      };
    }
    return {
      x: e.clientX - rect.left,
      y: e.clientY - rect.top,
    };
  };

  const scratch = (x: number, y: number) => {
    const ctx = contextRef.current;
    if (!ctx) return;

    ctx.globalCompositeOperation = 'destination-out';
    ctx.beginPath();
    ctx.arc(x, y, 30, 0, Math.PI * 2); // Radius 30 für guten Effekt
    ctx.fill();
    
    // Haptisches Feedback (Vibration) beim Rubbeln, wenn unterstützt
    if (navigator.vibrate) {
      navigator.vibrate(10);
    }
  };

  const handleStart = (e: React.MouseEvent | React.TouchEvent) => {
    e.preventDefault();
    setIsDrawing(true);
    const { x, y } = getCoords(e);
    scratch(x, y);
  };

  const handleMove = (e: React.MouseEvent | React.TouchEvent) => {
    if (!isDrawing) return;
    e.preventDefault();
    const { x, y } = getCoords(e);
    scratch(x, y);
  };

  const handleEnd = () => {
    setIsDrawing(false);
    // Hier könntest du prüfen, wie viel % freigerubbelt wurde.
    // Für MVP: Einfach einen "Enthüllen" Button nach dem ersten Rubbeln einblenden.
  };

  return (
    <div className="relative rounded-xl overflow-hidden shadow-2xl border border-gray-200" style={{ width, height }}>
      
      {/* Der versteckte Inhalt (Der Secret Spot) */}
      <div className="absolute inset-0 bg-white z-0 flex flex-col items-center justify-center p-6 text-center">
        {children}
      </div>

      {/* Das Canvas Overlay (Die Goldfolie) */}
      {!isRevealed && (
        <canvas
          ref={canvasRef}
          width={width}
          height={height}
          className="absolute inset-0 z-10 touch-none cursor-crosshair"
          onMouseDown={handleStart}
          onMouseMove={handleMove}
          onMouseUp={handleEnd}
          onMouseLeave={handleEnd}
          onTouchStart={handleStart}
          onTouchMove={handleMove}
          onTouchEnd={handleEnd}
        />
      )}

      {/* Optional: Button zum vollständigen Enthüllen (UX-Fallback) */}
      {!isRevealed && (
        <button 
          onClick={() => setIsRevealed(true)}
          className="absolute bottom-4 left-1/2 -translate-x-1/2 z-20 bg-black/70 text-white text-xs px-4 py-2 rounded-full backdrop-blur-md hover:bg-black transition"
        >
          Alles Enthüllen 🔓
        </button>
      )}
    </div>
  );
};

export default ScratchCard;
```

### Teil 3: Die Logik für den "Catch" (Backend/Datenbank)

Damit das System fair bleibt und du später die "Count System" Logik einbauen kannst, brauchst du eine saubere Datenbank-Struktur (z.B. in Supabase, Firebase oder Prisma).

**1. Tabelle: `spots`**
* `id` (UUID)
* `title` (String)
* `description` (Text)
* `local_tip` (Text)
* `latitude`, `longitude` (Float)
* `created_by` (User ID des Locals)
* `catch_radius` (Integer, z.B. 200 Meter)
* `requires_photo_proof` (Boolean)

**2. Tabelle: `catches` (Das Herzstück des Count Systems)**
* `id` (UUID)
* `spot_id` (Foreign Key zu spots)
* `user_id` (Foreign Key zum Reisenden)
* `caught_at` (Timestamp)
* `rating` (1-5 Sterne)
* `photo_proof_url` (Optional, wenn hochgeladen)
* `status` ('pending', 'verified')

**Die Logik für den "Ich bin vor Ort"-Button (MVP):**
1. Reisender klickt auf "Ich bin hier".
2. Frontend fragt die Browser-Geolocation ab (`navigator.geolocation.getCurrentPosition`).
3. Berechne die Distanz zwischen User-Position und Spot-Koordinaten (Haversine-Formel).
4. **Wenn Distanz < 200m:** Aktiviere die `ScratchCard`.
5. **Wenn Distanz > 200m:** Zeige Fehlermeldung: "Du bist noch zu weit weg! (X km entfernt)".
6. **Nach dem Freirubbeln:** Sende einen POST-Request an deine API, um den Eintrag in der `catches` Tabelle zu erstellen und dem Local eine Benachrichtigung zu schicken.

### Zusammenfassung der nächsten Schritte für dich:
1. **Datenbank aufsetzen:** Erstelle die Tabellen `spots` und `catches`.
2. **Komponente einbauen:** Kopiere den `ScratchCard` Code in dein Next.js Projekt.
3. **Erste Locals anschreiben:** Nutze die Vorlagen aus Teil 1, um die ersten 10 Spots manuell zu sammeln.
4. **Testen:** Geh selbst raus, setze einen Dummy-Spot in deiner Stadt und teste das "Catchen" mit dem Handy.
